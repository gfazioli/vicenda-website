# Release Notes

What changed in each version of Vicenda. The notes come straight from GitHub
Releases — the same feed the app's own updater reads — and refresh with every
release, so this page needs no editing when a version ships.

import { ReleaseNotes } from '@/components/ReleaseNotes/ReleaseNotes';
import { loadReleases } from '@/components/ReleaseNotes/load-releases';

{/*
  Fetched and compiled at build time, so the releases are in the served HTML:
  fetched in the browser, Googlebot got a 403 from the API route and saw only the
  loading skeleton. One fetch feeds both the page and its table of contents.
*/}
export const releases = await loadReleases();

<ReleaseNotes initialReleases={releases} />

export const metadata = {
    description:
        "What's new in Vicenda, pulled from GitHub Releases — the same feed the app's updater reads.",
    toc: releases.map((release) => ({
        value: `${release.tag_name} - ${new Date(release.created_at).toLocaleDateString('en-US', { month: 'short', year: 'numeric' }).replace(' ', ', ')}`,
        depth: '3',
        id: release.tag_name,
    })),
}
